! ===== ACC-PLACEMENT (Cisco 2960-24TT access switch) =====
! Uplinks: G0/1 -> CORE-1 Fa0/2 , G0/2 -> CORE-2 Fa0/2
enable
configure terminal
hostname ACC-PLACEMENT
no ip domain-lookup
ip domain-name placement.srm.edu
service password-encryption
service timestamps log datetime msec
enable secret Acc@2026
username localadmin privilege 15 secret Local@2026
banner motd #*** PlaceGuard Access Switch - Authorised access only ***#
spanning-tree mode rapid-pvst
!
vlan 10
 name STUDENTS
vlan 20
 name PLACEMENT
vlan 30
 name INTERVIEW
vlan 40
 name RECRUITER-WIFI
vlan 60
 name IOT
vlan 99
 name MGMT
vlan 998
 name PARKING
vlan 999
 name NATIVE
exit
!
! ---- DHCP snooping (blocks rogue DHCP servers) ----
ip dhcp snooping
ip dhcp snooping vlan 10,20,30,40,60
no ip dhcp snooping information option
!
! ---- Uplink trunks (trusted) ----
interface range GigabitEthernet0/1 - 2
 description Trunk uplinks to CORE-1 / CORE-2
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,40,60,99,999
 switchport nonegotiate
 ip dhcp snooping trust
 no shutdown
exit

 description Placement Cell PCs
 switchport mode access
 switchport access vlan 20
 switchport port-security
 switchport port-security maximum 1

 switchport port-security violation shutdown
 spanning-tree portfast
 spanning-tree bpduguard enable
 ip dhcp snooping limit rate 10
 no shutdown
exit
interface FastEthernet0/5
 description Network admin PC
