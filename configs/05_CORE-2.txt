! ---- Role-based inter-VLAN access control (applied inbound on the SVIs) ----
ip access-list extended STUDENT_IN
 permit udp any any eq bootps
 permit icmp 10.10.10.0 0.0.0.255 host 10.10.10.1
 deny ip 10.10.10.0 0.0.0.255 10.10.0.0 0.0.255.255
 permit ip 10.10.10.0 0.0.0.255 any
exit
ip access-list extended INTERVIEW_IN
 permit udp any any eq bootps
 deny ip 10.10.30.0 0.0.0.255 10.10.10.0 0.0.0.255
 deny ip 10.10.30.0 0.0.0.255 10.10.60.0 0.0.0.255
 deny ip 10.10.30.0 0.0.0.255 10.10.99.0 0.0.0.255
 permit ip any any
exit
ip access-list extended WIFI_IN
 permit udp any any eq bootps
 permit icmp 10.10.40.0 0.0.0.255 host 10.10.40.1
 permit ip 10.10.40.0 0.0.0.255 10.10.30.0 0.0.0.255
 deny ip 10.10.40.0 0.0.0.255 10.10.0.0 0.0.255.255
 permit ip 10.10.40.0 0.0.0.255 any
exit
ip access-list extended IOT_IN
 ! (added after PT test) let the other core's HSRPv2 hellos in, otherwise both cores go Active for VLAN 60
 permit udp any host 224.0.0.102 eq 1985
 permit udp any any eq bootps
 permit tcp 10.10.60.0 0.0.0.255 10.10.20.0 0.0.0.255 established
 permit tcp 10.10.60.0 0.0.0.255 10.10.99.0 0.0.0.255 established
 permit icmp 10.10.60.0 0.0.0.255 10.10.20.0 0.0.0.255 echo-reply
