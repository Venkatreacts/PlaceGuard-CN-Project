! ===== FW-DC (Cisco ASA 5506-X) - Data-centre firewall: CAMPUS / SERVERS / DMZ =====
! When asked for the enable password the first time, just press Enter.
! (Tested in PT 9.0.1: the PT 5506-X starts with no DHCP server, so the old "no dhcpd" lines were removed.)
enable
configure terminal
hostname FW-DC
enable password Asa@2026
domain-name placement.srm.edu
!
interface GigabitEthernet1/1
 description To SW-TRANSIT (campus core, VLAN 100)
 nameif CAMPUS
 security-level 50
 ip address 10.10.100.4 255.255.255.248
 no shutdown
interface GigabitEthernet1/2
 description To SW-SERVERS (internal server farm)
 nameif SERVERS
 security-level 100
 ip address 172.16.50.1 255.255.255.0
 no shutdown
interface GigabitEthernet1/3
 description To PORTAL-SRV (DMZ)
 nameif DMZ
 security-level 70
 ip address 172.16.70.1 255.255.255.0
 no shutdown
exit
!
route CAMPUS 0.0.0.0 0.0.0.0 10.10.100.1
!
! ---- What the campus (and recruiter HQ via VPN, and Internet via NAT) may reach ----
! infrastructure services
access-list CAMPUS_IN extended permit udp 10.10.0.0 255.255.0.0 host 172.16.50.10 eq 67
access-list CAMPUS_IN extended permit udp 10.10.0.0 255.255.0.0 host 172.16.50.10 eq 53
access-list CAMPUS_IN extended permit udp 192.168.50.0 255.255.255.0 host 172.16.50.10 eq 53
access-list CAMPUS_IN extended permit udp 10.10.0.0 255.255.0.0 host 172.16.50.12 eq 514
access-list CAMPUS_IN extended permit udp 10.10.0.0 255.255.0.0 host 172.16.50.12 eq 123
access-list CAMPUS_IN extended permit tcp 10.10.0.0 255.255.0.0 host 172.16.50.11 eq 49
! Placement Cell and Network Admins: full access (administrative role)
access-list CAMPUS_IN extended permit ip 10.10.20.0 255.255.255.0 any
access-list CAMPUS_IN extended permit ip 10.10.99.0 255.255.255.0 any
! Placement portal (DMZ): web for everyone, resume FTP for students
access-list CAMPUS_IN extended permit tcp any host 172.16.70.10 eq 80
access-list CAMPUS_IN extended permit tcp any host 172.16.70.10 eq 443
access-list CAMPUS_IN extended permit tcp 10.10.10.0 255.255.255.0 host 172.16.70.10 eq 21
access-list CAMPUS_IN extended permit icmp 10.10.0.0 255.255.0.0 host 172.16.70.10
! Offer-letter e-mail
access-list CAMPUS_IN extended permit tcp 10.10.0.0 255.255.0.0 host 172.16.50.13 eq 25
access-list CAMPUS_IN extended permit tcp 10.10.0.0 255.255.0.0 host 172.16.50.13 eq 110
access-list CAMPUS_IN extended permit tcp 192.168.50.0 255.255.255.0 host 172.16.50.13 eq 25
access-list CAMPUS_IN extended permit tcp 192.168.50.0 255.255.255.0 host 172.16.50.13 eq 110
! everything else (e.g. students -> student-records DB) is dropped
access-list CAMPUS_IN extended deny ip any any
access-group CAMPUS_IN in interface CAMPUS
!
! ---- DMZ: the portal may only query the DB and DNS; it can never start a connection to the campus ----
access-list DMZ_IN extended permit tcp host 172.16.70.10 host 172.16.50.14 eq 80
access-list DMZ_IN extended permit udp host 172.16.70.10 host 172.16.50.10 eq 53
! (added after PT test) PT's ASA checks the portal's REPLIES against DMZ_IN, so without these
! lines every ping/web/FTP reply from the portal was dropped (deny counter rose, T1 failed).
! They only allow traffic coming FROM the portal's service ports, so the portal still cannot open new connections.
access-list DMZ_IN extended permit icmp host 172.16.70.10 any echo-reply
access-list DMZ_IN extended permit tcp host 172.16.70.10 eq www any
access-list DMZ_IN extended permit tcp host 172.16.70.10 eq 443 any
access-list DMZ_IN extended permit tcp host 172.16.70.10 eq ftp any
access-list DMZ_IN extended deny ip any any
access-group DMZ_IN in interface DMZ
!
! ---- Stateful inspection so replies (incl. ping replies) are allowed back ----
class-map inspection_default
 match default-inspection-traffic
exit
policy-map global_policy
 class inspection_default
  inspect icmp
  inspect dns
  inspect ftp
  inspect http
! On PT's ASA one "exit" from class level returns straight to (config)
exit
! PT already applies global_policy by default; this line just confirms it (shows a WARNING)
service-policy global_policy global
!
! ---- Syslog: PT 9.0.1's ASA has no "logging" command ("% Unrecognized command"), so the ASA
! ---- cannot send logs. Routers and switches still log to LOG-NTP-SRV 172.16.50.12.
end
write memory
